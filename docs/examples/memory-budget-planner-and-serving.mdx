## Tight-VRAM Execution + Structured Logs + FastAPI

```python
import json
from pathlib import Path
import numpy as np
from fastapi.testclient import TestClient

from kitepy import CNN

log_path = Path("./logs/events.jsonl")
bundle_path = Path("./artifacts/resnet50_safe.safetensors")

model = CNN("resnet50", num_classes=10)
model.train(
    "cifar10",
    epochs=2,
    batch_size=64,
    auto_plan=True,
    memory_budget_mb=380.0,         # Automatically solves micro_bs & grad_accum
    auto_recover=True,
    structured_log_file=str(log_path),
    safe_serialization=True,
    save_every_n_epochs=0,
)

# Inspect solved plan, recovery stats, and lineage
print(json.dumps(model.execution_report, indent=2))

# Save pickle-free SafeTensors bundle + reload cleanly
model.save(bundle_path, safe=True)
reloaded = CNN.load(bundle_path)

# Serve live inference endpoint
client = TestClient(reloaded.to_fastapi())
print("Health:", client.get("/health").json())
print("Predict:", client.post("/predict", json={"inputs": np.random.randn(1, 3, 32, 32).tolist(), "top_k": 3}).json())
```
