## Custom `nn.Module` + SafeTensors Save/Load

```python
import torch
import torch.nn as nn
from kitepy import CNN

class SmallNet(nn.Module):
    def __init__(self, num_classes=3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Conv2d(3, 16, 3, padding=1),
            nn.ReLU(),
            nn.AdaptiveAvgPool2d(1),
            nn.Flatten(),
            nn.Linear(16, num_classes),
        )

    def forward(self, x):
        return self.net(x)

model = CNN(SmallNet(num_classes=3))
model.train("synthetic", epochs=2, img_size=32, batch_size=8, save_every_n_epochs=0)
model.save("custom_net.safetensors", safe=True)

# Custom architectures require passing custom_model= on reload
reloaded = CNN.load("custom_net.safetensors", custom_model=SmallNet(num_classes=3))
print(reloaded.predict(torch.randn(1, 3, 32, 32), top_k=3))
```

## Custom `train_step` Hook

Keep KitePy's AMP, gradient accumulation, callbacks, EMA, and recovery while overriding the batch loss step:

```python
import torch.nn.functional as F

def focal_style_step(engine, batch):
    x, y = batch
    logits = engine.model(x)
    loss = F.cross_entropy(logits, y, label_smoothing=0.1)
    return loss, logits, y

CNN("resnet18").train("synthetic", epochs=1, img_size=32, train_step=focal_style_step)
```
